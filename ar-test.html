const model = document.getElementById( "arizona-model-3d" ); model.addEventListener( "model-loaded", () => { console.log( "✅ GLB cargado correctamente" ); status.textContent = "✅ Modelo cargado — busca el marcador"; const object = model.getObject3D("mesh"); if (!object) { console.error( "No se encontró el objeto 3D interno" ); return; } // ===================================================== // CALCULAR DIMENSIONES REALES DEL MODELO // ===================================================== const box = new THREE.Box3().setFromObject(object); const size = new THREE.Vector3(); const center = new THREE.Vector3(); box.getSize(size); box.getCenter(center); console.log( "Tamaño original:", size ); console.log( "Centro original:", center ); // ===================================================== // CENTRAR EL MODELO // ===================================================== object.position.x -= center.x; // Los pies quedan apoyados aproximadamente en Y = 0 object.position.y -= box.min.y; object.position.z -= center.z; // ===================================================== // ESCALAR AUTOMÁTICAMENTE // ===================================================== const alturaDeseada = 0.8; const escala = alturaDeseada / size.y; model.object3D.scale.set( escala, escala, escala ); console.log( "Escala aplicada:", escala ); console.log( "✅ Modelo centrado automáticamente" ); } ); model.addEventListener( "model-error", event => { console.error( "❌ ERROR CARGANDO GLB", event ); status.textContent = "❌ El archivo GLB no pudo cargarse"; } );
